import * as THREE from 'three';
import type { Island3DQuality } from './island5ThreePilotContract';
import type { Island2CelestialMaterials } from './Island2CelestialThreeWorld';

type PalaceLevel = 1 | 2 | 3;

/** Palace-local coordinates: +Y up, +Z arrival stair; the world owns placement. */
export function createCelestialV2Palace(
  level: PalaceLevel,
  quality: Island3DQuality,
  materials: Island2CelestialMaterials,
): THREE.Group {
  const root = new THREE.Group();
  root.name = `ISLAND_2_SOLSPIRE_PALACE_V2_L${level}`;
  const radialSegments = quality === 'high' ? 24 : quality === 'medium' ? 16 : 12;

  function phase(buildLevel: PalaceLevel) {
    const group = new THREE.Group();
    group.name = `SOLSPIRE_FUNDED_L${buildLevel}`;
    group.userData.palaceBuildLevel = buildLevel;
    root.add(group);
    return group;
  }

  function mesh(
    parent: THREE.Group,
    name: string,
    geometry: THREE.BufferGeometry,
    material: THREE.Material,
    position: [number, number, number],
    stage = 2,
  ) {
    const result = new THREE.Mesh(geometry, material);
    result.name = name;
    result.position.set(...position);
    result.castShadow = quality === 'high';
    result.receiveShadow = true;
    result.userData.constructionStage = stage;
    parent.add(result);
    return result;
  }

  function block(parent: THREE.Group, name: string, size: [number, number, number], at: [number, number, number], material = materials.ivory, stage = 2) {
    return mesh(parent, name, new THREE.BoxGeometry(...size), material, at, stage);
  }

  function drum(parent: THREE.Group, name: string, radius: number, height: number, at: [number, number, number], material = materials.ivory, stage = 2, sides = radialSegments) {
    return mesh(parent, name, new THREE.CylinderGeometry(radius, radius, height, sides), material, at, stage);
  }

  // Closed extruded masonry, with a genuine traversable opening rather than a
  // dark rectangle pasted onto a solid wall. Its curved underside reads in orbit.
  function arcade(parent: THREE.Group, name: string, width: number, height: number, depth: number, at: [number, number, number], rotationY = 0, pointedGable = false) {
    const pier = width * 0.17;
    const spring = height * 0.58;
    const outer = new THREE.Shape();
    outer.moveTo(-width / 2, 0);
    if (pointedGable) {
      outer.lineTo(-width / 2, height * 0.90);
      outer.lineTo(0, height * 1.29);
      outer.lineTo(width / 2, height * 0.90);
    } else {
      outer.lineTo(-width / 2, height);
      outer.lineTo(width / 2, height);
    }
    outer.lineTo(width / 2, 0);
    outer.lineTo(width / 2 - pier, 0);
    outer.lineTo(width / 2 - pier, spring);
    outer.quadraticCurveTo(width * 0.30, height * 0.80, 0, height * 0.91);
    outer.quadraticCurveTo(-width * 0.30, height * 0.80, -width / 2 + pier, spring);
    outer.lineTo(-width / 2 + pier, 0);
    outer.closePath();
    const arch = mesh(parent, name, new THREE.ExtrudeGeometry(outer, {
      depth, bevelEnabled: false, curveSegments: quality === 'low' ? 5 : 8,
    }), materials.ivory, at);
    arch.rotation.y = rotationY;
    return arch;
  }

  function pitchedRoof(parent: THREE.Group, name: string, width: number, height: number, depth: number, at: [number, number, number]) {
    const profile = new THREE.Shape();
    profile.moveTo(-width / 2, 0);
    profile.lineTo(0, height);
    profile.lineTo(width / 2, 0);
    profile.closePath();
    return mesh(parent, name, new THREE.ExtrudeGeometry(profile, { depth, bevelEnabled: false }), materials.sapphire, at, 3);
  }

  function roofSlope(parent: THREE.Group, name: string, side: number) {
    const profile = new THREE.Shape();
    profile.moveTo(side * 0.36, 0);
    profile.lineTo(side * 0.36, 0.29);
    profile.lineTo(side * 0.76, 0.04);
    profile.lineTo(side * 0.76, 0);
    profile.closePath();
    return mesh(parent, name, new THREE.ExtrudeGeometry(profile, { depth: 0.72, bevelEnabled: false }), materials.sapphire, [0, 1.415, -0.46], 3);
  }

  function roof(parent: THREE.Group, name: string, radius: number, height: number, at: [number, number, number]) {
    // A flared eave and inward-bowed steep taper give the source its long,
    // architectural spires; a hemisphere or blunt toy cone loses that identity.
    const profile = [
      [0, 0], [radius, 0], [radius, height * 0.035],
      [radius * 0.89, height * 0.09], [radius * 0.72, height * 0.27],
      [radius * 0.48, height * 0.52], [radius * 0.24, height * 0.77],
      [radius * 0.055, height * 0.95], [0, height],
    ].map(([r, y]) => new THREE.Vector2(r, y));
    return mesh(parent, name, new THREE.LatheGeometry(profile, radialSegments), materials.sapphire, at, 3);
  }

  function spire(parent: THREE.Group, name: string, x: number, z: number, bottom: number, bodyHeight: number, radius: number, roofHeight: number) {
    const group = new THREE.Group();
    group.name = name;
    group.position.set(x, bottom, z);
    parent.add(group);
    drum(group, `${name}_FOOT`, radius * 1.12, 0.12, [0, 0.06, 0], materials.ivoryShade, 1, 8);
    drum(group, `${name}_SHAFT`, radius, bodyHeight, [0, bodyHeight / 2, 0], materials.ivory, 2, 8);
    drum(group, `${name}_CAPITAL`, radius * 1.10, 0.11, [0, bodyHeight - 0.025, 0], materials.ivory, 3, 8);
    roof(group, `${name}_SAPPHIRE_ROOF`, radius * 1.19, roofHeight, [0, bodyHeight + 0.015, 0]);
    return group;
  }

  const first = phase(1);
  drum(first, 'PALACE_LOWER_COURT', 1.77, 0.14, [0, 0.07, 0], materials.ivoryShade, 1);

  // The raised terrace has a physical notch for the arrival stair. Retaining
  // walls continue around the rear and sides; no front-only scenery shells.
  const terrace = new THREE.Shape();
  const notchAngle = 0.36;
  for (let i = 0; i <= 40; i += 1) {
    const angle = notchAngle + (Math.PI * 2 - notchAngle * 2) * i / 40;
    const x = Math.sin(angle) * 1.6;
    const z = Math.cos(angle) * 1.6;
    if (i === 0) terrace.moveTo(x, -z);
    else terrace.lineTo(x, -z);
  }
  terrace.lineTo(-Math.sin(notchAngle) * 1.6, -0.70);
  terrace.lineTo(Math.sin(notchAngle) * 1.6, -0.70);
  terrace.closePath();
  const terraceGeometry = new THREE.ExtrudeGeometry(terrace, { depth: 0.31, bevelEnabled: false });
  terraceGeometry.rotateX(-Math.PI / 2);
  mesh(first, 'PALACE_NOTCHED_UPPER_TERRACE', terraceGeometry, materials.ivory, [0, 0.14, 0], 1);
  for (let step = 0; step < 7; step += 1) {
    const front = 1.70 - step * 0.125;
    const height = 0.045 * (step + 1);
    block(first, `PALACE_ARRIVAL_STEP_${step}`, [1.04, height, front - 0.69], [0, 0.14 + height / 2, (front + 0.69) / 2], materials.ivory, 1);
  }

  block(first, 'PALACE_LOWER_HALL', [1.34, 0.95, 1.36], [0, 0.925, -0.13]);
  // Four pitched roof volumes surround the central lantern socket. This keeps
  // the complete L1 roof intact when the later crown rises through its opening.
  pitchedRoof(first, 'PALACE_NAVE_FRONT_ROOF', 1.52, 0.47, 0.48, [0, 1.415, 0.26]);
  pitchedRoof(first, 'PALACE_NAVE_REAR_ROOF', 1.52, 0.47, 0.41, [0, 1.415, -0.87]);
  roofSlope(first, 'PALACE_NAVE_WEST_ROOF', -1);
  roofSlope(first, 'PALACE_NAVE_EAST_ROOF', 1);
  arcade(first, 'PALACE_DEEP_ENTRY', 0.94, 1.11, 0.36, [0, 0.45, 0.57], 0, true);
  block(first, 'PALACE_ENTRY_RECESSED_DOORS', [0.48, 0.77, 0.05], [0, 0.84, 0.567], materials.sapphire, 4);
  for (const side of [-1, 1]) {
    // Faceted piers support the nave instead of a flat gatehouse lintel.
    const pier = new THREE.Shape();
    pier.moveTo(-0.11, 0);
    pier.lineTo(-0.11, 0.98);
    pier.lineTo(0, 1.37);
    pier.lineTo(0.11, 0.98);
    pier.lineTo(0.11, 0);
    pier.closePath();
    mesh(first, `PALACE_PORTAL_CLUSTERED_PIER_${side}`, new THREE.ExtrudeGeometry(pier, { depth: 0.39, bevelEnabled: false }), materials.ivory, [side * 0.53, 0.45, 0.53]);
  }
  // Low corner towers belong to the first funded building forever.
  spire(first, 'PALACE_FRONT_WEST_SPIRE', -1.16, 0.46, 0.45, 1.11, 0.185, 0.58);
  spire(first, 'PALACE_FRONT_EAST_SPIRE', 1.16, 0.46, 0.45, 1.19, 0.185, 0.61);

  if (level >= 2) {
    const second = phase(2);
    for (const side of [-1, 1]) {
      const label = side < 0 ? 'WEST' : 'EAST';
      const x = side * 1.02;
      block(second, `PALACE_${label}_GALLERY_FLOOR`, [0.75, 0.12, 1.20], [x, 0.51, -0.22], materials.ivoryShade, 1);
      block(second, `PALACE_${label}_GALLERY_VAULT`, [0.75, 0.16, 1.20], [x, 1.14, -0.22]);
      pitchedRoof(second, `PALACE_${label}_GALLERY_ROOF`, 0.82, 0.29, 1.26, [x, 1.22, -0.85]);
      // The gallery has open front/rear and side bays, visibly connected to
      // the central hall through its slab rather than floating beside it.
      arcade(second, `PALACE_${label}_GALLERY_FRONT`, 0.75, 0.60, 0.12, [x, 0.57, 0.30]);
      arcade(second, `PALACE_${label}_GALLERY_REAR`, 0.75, 0.60, 0.12, [x, 0.57, -0.82]);
      for (let bay = 0; bay < 2; bay += 1) {
        arcade(second, `PALACE_${label}_GALLERY_SIDE_${bay}`, 0.51, 0.60, 0.12, [side * 1.34, 0.57, -0.51 + bay * 0.51], Math.PI / 2);
      }
      spire(second, `PALACE_${label}_HIGH_SPIRE`, side * 0.73, -0.42, 0.45, side < 0 ? 1.84 : 1.73, 0.205, side < 0 ? 0.73 : 0.69);
    }
    block(second, 'PALACE_REAR_GALLERY_FLOOR', [1.38, 0.12, 0.44], [0, 0.51, -1.04], materials.ivoryShade, 1);
    block(second, 'PALACE_REAR_GALLERY_VAULT', [1.38, 0.13, 0.44], [0, 1.15, -1.04]);
    pitchedRoof(second, 'PALACE_REAR_GALLERY_ROOF', 1.42, 0.30, 0.47, [0, 1.22, -1.28]);
    for (let bay = -1; bay <= 1; bay += 1) {
      arcade(second, `PALACE_REAR_GALLERY_BAY_${bay + 1}`, 0.46, 0.60, 0.12, [bay * 0.46, 0.57, -1.27]);
    }
  }

  if (level >= 3) {
    const third = phase(3);
    // A tall, narrow lantern replaces the first blockout's top-heavy solid tube.
    // Its eight structural arch bays and recessed core give the upper masonry
    // genuine depth, including the rear. The crown fits the original roof socket.
    const lantern = new THREE.Group();
    lantern.name = 'PALACE_PRINCIPAL_SPIRE';
    lantern.position.set(0, 1.40, -0.10);
    third.add(lantern);
    drum(lantern, 'PALACE_PRINCIPAL_LANTERN_FOOT', 0.35, 0.72, [0, 0.36, 0], materials.ivory, 2, 8);
    drum(lantern, 'PALACE_PRINCIPAL_LANTERN_SILL', 0.37, 0.12, [0, 0.72, 0], materials.ivoryShade, 2, 8);
    drum(lantern, 'PALACE_PRINCIPAL_RECESSED_CHAMBER', 0.22, 0.69, [0, 1.10, 0], materials.sapphire, 2, 8);
    for (let bay = 0; bay < 8; bay += 1) {
      const angle = bay / 8 * Math.PI * 2;
      arcade(lantern, `PALACE_PRINCIPAL_LANTERN_BAY_${bay}`, 0.275, 0.81, 0.07,
        [Math.sin(angle) * 0.318, 0.77, Math.cos(angle) * 0.318], angle);
    }
    drum(lantern, 'PALACE_PRINCIPAL_LANTERN_CROWN', 0.37, 0.12, [0, 1.57, 0], materials.ivory, 3, 8);
    roof(lantern, 'PALACE_PRINCIPAL_SAPPHIRE_ROOF', 0.405, 1.12, [0, 1.625, 0]);
    for (const side of [-1, 1]) {
      spire(third, `PALACE_${side < 0 ? 'WEST' : 'EAST'}_REAR_SPIRE`, side * 1.07, -0.90, 0.45, side < 0 ? 1.27 : 1.35, 0.18, 0.62);
    }
  }

  return root;
}
